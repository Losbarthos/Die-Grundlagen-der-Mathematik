\chapter{Ergänzende Bildrechnungen}
\section{Bilder kleiner aufgezählter Mengen}
Diese Rechenregeln bündeln das Bildgesetz für Vereinigungen und das
bereits bewiesene Bildgesetz für Einermengen.

\Needspace{14\baselineskip}
\FormulaThmDeltaK[Bild einer Zweiermenge]{%
f\colon A\to B\dsep a,b\in A\vdash f[\{a,b\}]=\{f(a),f(b)\}
}{FunctionImagePair}{%
  \DeltaRow{Mengen}{A\dsep B\dsep a\dsep b}
  \DeltaRow{Funktionen}{f}
}
\noindent Die Funktion \(f\) ist ein Parameter der Aussage.
\begin{tabproofwide}
  \proofstepwidestar[1]{f\colon A\to B}{\rA}
  \proofstepwidestar[2]{a,b\in A}{\rA}
  \proofstepwidestar[2]{\{a\}\subseteq A}{\FormulaRefAuto{a\in A\vdash\{a\}\subseteq A}[thm]{\rAEa{2}}}
  \proofstepwidestar[2]{\{b\}\subseteq A}{\FormulaRefAuto{a\in A\vdash\{a\}\subseteq A}[thm]{\rAEb{2}}}
  \proofstepwidestar[1,2]{f[\{a\}\cup\{b\}]=f[\{a\}]\cup f[\{b\}]}{\FormulaRefAuto{F\colon M\to N\dsep A\subseteq M\dsep B\subseteq M\vdash F[A\cup B]=F[A]\cup F[B]}[thm]{1,3,4}}
  \proofstepwidestar[1,2]{f[\{a\}]=\{f(a)\}}{\FormulaRefAuto{x\in A\dsep F\colon A\to B\vdash F[\{x\}]=\{F(x)\}}[thm]{\rAEa{2},1}}
  \proofstepwidestar[1,2]{f[\{b\}]=\{f(b)\}}{\FormulaRefAuto{x\in A\dsep F\colon A\to B\vdash F[\{x\}]=\{F(x)\}}[thm]{\rAEb{2},1}}
  \proofstepwidestar[]{\{a,b\}=\{a\}\cup\{b\}}{\FormulaRefAuto{\{a,b\}=\{a\}\cup\{b\}}[thm]}
  \proofstepwidestar[]{\{f(a),f(b)\}=\{f(a)\}\cup\{f(b)\}}{\FormulaRefAuto{\{a,b\}=\{a\}\cup\{b\}}[thm]}
  \proofstepwidestar[1,2]{f[\{a,b\}]=\{f(a),f(b)\}}{\rIE{\FormulaRefAuto{a=b\vdash b=a}[thm]{8},6,7,\FormulaRefAuto{a=b\vdash b=a}[thm]{9},5}}
\end{tabproofwide}
\Needspace{14\baselineskip}
\FormulaThmDeltaK[Bild einer Dreiermenge]{%
f\colon A\to B\dsep a,b,c\in A\vdash f[\{a,b,c\}]=\{f(a),f(b),f(c)\}
}{FunctionImageTriple}{%
  \DeltaRow{Mengen}{A\dsep B\dsep a\dsep b\dsep c}
  \DeltaRow{Funktionen}{f}
}
\noindent Wieder bezeichnet \(f\) den Funktionsparameter.
\begin{tabproofwide}
  \proofstepwidestar[1]{f\colon A\to B}{\rA}
  \proofstepwidestar[2]{a,b,c\in A}{\rA}
  \proofstepwidestar[2]{\{a,b\}\subseteq A}{\FormulaRefAuto{PairSetSubsetFromMembership}[thm]{\rAEn{2},\rAEn{2}}}
  \proofstepwidestar[2]{\{c\}\subseteq A}{\FormulaRefAuto{a\in A\vdash\{a\}\subseteq A}[thm]{\rAEn{2}}}
  \proofstepwidestar[1,2]{f[\{a,b\}\cup\{c\}]=f[\{a,b\}]\cup f[\{c\}]}{\FormulaRefAuto{F\colon M\to N\dsep A\subseteq M\dsep B\subseteq M\vdash F[A\cup B]=F[A]\cup F[B]}[thm]{1,3,4}}
  \proofstepwidestar[1,2]{f[\{a,b\}]=\{f(a),f(b)\}}{\FormulaRefAuto{FunctionImagePair}[thm]{1,\rAI{\rAEn{2},\rAEn{2}}}}
  \proofstepwidestar[1,2]{f[\{c\}]=\{f(c)\}}{\FormulaRefAuto{x\in A\dsep F\colon A\to B\vdash F[\{x\}]=\{F(x)\}}[thm]{\rAEn{2},1}}
  \proofstepwidestar[]{\{a,b,c\}=\{a,b\}\cup\{c\}}{\FormulaRefAuto{TripleSetDef}[def]}
  \proofstepwidestar[]{\{f(a),f(b),f(c)\}=\{f(a),f(b)\}\cup\{f(c)\}}{\FormulaRefAuto{TripleSetDef}[def]}
  \proofstepwidestar[1,2]{f[\{a,b,c\}]=\{f(a),f(b),f(c)\}}{\rIE{\FormulaRefAuto{a=b\vdash b=a}[thm]{8},6,7,\FormulaRefAuto{a=b\vdash b=a}[thm]{9},5}}
\end{tabproofwide}
