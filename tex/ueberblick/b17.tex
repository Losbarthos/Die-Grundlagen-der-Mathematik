\chapter[Band 17: Ganze Zahlen]{Band 17 auf einen Blick}
\noindent\textbf{Ganze Zahlen}\par\medskip
\noindent\emph{Lesart.}
In Klassenformeln sind \(a,b,c,d\in\mathbb N\), in der Ganzzahlarithmetik
\(z,w,u\in\mathbb Z\). Die Zahlzeichen \(0_{\mathbb Z}=\IntZero\) und
\(1_{\mathbb Z}=\IntOne\) bezeichnen die eingebetteten natürlichen Zahlen.
Die Axiome werden im ausgelagerten Quotientenmodell bewiesen.
Die \IntegersReadingLink{} erklärt die vollständige Konstruktion;
die \IntegersProofsLink{} enthalten alle formalen Ableitungen.
Öffentliche Aussagen bleiben in Band~17, lokale Modellresultate tragen
den Präfix \(17\mathrm E\). Die Übersicht unterscheidet diese Rollen.
Die \DifferencesReadingLink{} zu Band~41 verallgemeinert das additive
Prinzip zum Einbettungssatz für kommutative kürzbare Monoide;
die zugehörigen \DifferencesProofsLink{} enthalten dessen formale Konstruktion.

\begin{BandUebersicht}
\textbf{Differenzenpaare und ihre Äquivalenz}
\UebFormel{\begin{aligned}
D_{\mathbb Z}&\coloneqq\mathbb N\times\mathbb N,\\
(a,b)\IntPairRel(c,d)&\coloneqq a+d=b+c.
\end{aligned}}
\UebQuellen{\IntegersResultRef{IntegerPairCarrier}[def];
\IntegersResultRef{IntPairRelDef}[def]}
&
\textit{Äquivalenzrelation}
\UebFormel{\EqRel{D_{\mathbb Z},\IntPairRel}.}
Reflexivität, Symmetrie und Transitivität folgen aus der natürlichen
Addition und ihrer Kürzbarkeit.
\UebQuellen{\IntegersResultRef{IntegerPairEquivalence}[thm]}
\\ \addlinespace[.8ex]

\textbf{Quotient und Ganzzahlklassen}
\UebFormel{\begin{aligned}
\mathbb Z&\coloneqq\QuotientSet{D_{\mathbb Z}}{\IntPairRel},\\
\IntClass a b&\coloneqq
\EqClass{(a,b)}{\IntPairRel}{D_{\mathbb Z}}.
\end{aligned}}
\UebQuellen{\IntegersResultRef{IntegersAsQuotient}[def];
\IntegersResultRef{IntegerClassDef}[def]}
&
\textit{Repräsentanten und Gleichheitskriterium}
\UebFormel{\begin{aligned}
&\IntClass a b=\IntClass c d\\
&\qquad\leftrightarrow a+d=b+c,\\
&z\in\mathbb Z\vdash
\exists a,b\in\mathbb N\;(z=\IntClass a b).
\end{aligned}}
\UebQuellen{\IntegersResultRef{IntegerClassEquality}[thm];
\IntegersResultRef{IntegerPairRepresentative}[thm]}
\\ \addlinespace[.8ex]

\textbf{Natürliche Einbettung und Teilmengenkopie}
\UebFormel{\begin{aligned}
&\IntEmbed\colon\mathbb N\to\mathbb Z,\qquad
\IntEmbed(n)\coloneqq\IntClass n0,\\
&\NatCopyInInt\coloneqq\IntEmbed[\mathbb N].
\end{aligned}}
\UebQuellen{\IntegersResultRef{NaturalIntegerEmbedding}[def];
\IntegersResultRef{NaturalCopyInIntegers}[def]}
&
\textit{Injektivität und Verträglichkeit}
Für \(m,n\in\mathbb N\):
\UebFormel{\begin{aligned}
&\IntEmbed\colon\mathbb N\inj\mathbb Z,\\
&\IntEmbed(m+n)=\IntEmbed(m)+\IntEmbed(n),\\
&\IntEmbed(mn)=\IntEmbed(m)\IntEmbed(n).
\end{aligned}}
\UebQuellen{\IntegersResultRef{NaturalIntegerEmbeddingInjective}[thm];
\IntegersResultRef{NaturalIntegerEmbeddingAdditive}[thm];
\IntegersResultRef{NaturalIntegerEmbeddingMultiplicative}[thm]}
\\ \addlinespace[.8ex]

\textbf{Negation, Addition und Subtraktion}
\UebFormel{\begin{aligned}
-\IntClass a b&\coloneqq\IntClass b a,\\
\IntClass a b+\IntClass c d
&\coloneqq\IntClass{a+c}{b+d},\\
z-w&\coloneqq z+(-w).
\end{aligned}}
\UebQuellen{\IntegersResultRef{IntegerNegationDef}[def];
\IntegersResultRef{IntegerAdditionDef}[def];
\IntegersResultRef{IntegerSubtractionDef}[def]}
&
\textit{Repräsentantenunabhängigkeit und additive Gesetze}
\UebFormel{\begin{aligned}
&(z+w)+u=z+(w+u),\\
&z+w=w+z,\qquad z+\IntZero=z,\\
&z+(-z)=\IntZero,\qquad-(-z)=z.
\end{aligned}}
\UebQuellen{\IntegersResultRef{IntegerAdditionCompatible}[thm];
\IntegersResultRef{IntAddAssociative}[thm];
\IntegersResultRef{IntAddCommutative}[thm];
\IntegersResultRef{IntAddZero}[thm];
\IntegersResultRef{IntAddInverse}[thm];

\textbf{Multiplikation}

\textbf{Schritte um eins}
Als Abbildungen \(\mathbb Z\to\mathbb Z\) werden eingeführt:
\UebFormel{\begin{aligned}
&(x\mapsto x+\IntOne)(z)\coloneqq z+\IntOne,\\
&(x\mapsto x-\IntOne)(z)\coloneqq z-\IntOne.
\end{aligned}}
\UebQuellen{\IntegersResultRef{IntegerSuccessorDef}[def];
\IntegersResultRef{IntegerPredecessorDef}[def]}
&
\textit{Inverse Schritte und Normalform}
\UebFormel{\begin{aligned}
&(z+\IntOne)-\IntOne=z,\\
&(z-\IntOne)+\IntOne=z,\\
&\exists n\in\mathbb N\;
 \bigl(z=\IntEmbed(n)\lor z=-\IntEmbed(n)\bigr).
\end{aligned}}
Insbesondere ist der Schritt nach rechts bijektiv.
\UebQuellen{\IntegersResultRef{IntegerPredAfterSucc}[thm];
\IntegersResultRef{IntegerSuccAfterPred}[thm];
\IntegersResultRef{IntegerSuccessorBijective}[thm];
\IntegersResultRef{IntegerSignedNormalForm}[thm]}
\\ \addlinespace[.8ex]

\textbf{Quotientenordnung}
\UebFormel{\begin{aligned}
&\IntClass a b\leq_{\mathbb Z}\IntClass c d
\coloneqq a+d\leq c+b,\\
&z<w\coloneqq z\leq w\land z\neq w.
\end{aligned}}
\UebQuellen{\IntegersResultRef{IntegerOrderDef}[def];
\IntegersResultRef{IntegerOrderNotation}[def];
\IntegersResultRef{IntegerStrictOrderDef}[def]}
&
\textit{Totale und translationsverträgliche Ordnung}
\UebFormel{\begin{aligned}
&\TotOrd{\mathbb Z,\leq_{\mathbb Z}},\\
&z\leq w\leftrightarrow z+u\leq w+u,\\
&z<z+\IntOne.
\end{aligned}}
\UebQuellen{\IntegersResultRef{IntegerOrderTotal}[thm];
\IntegersResultRef{IntegerOrderTranslationModel}[thm];
\IntegersResultRef{IntegerSuccessorPositive}[thm]}
\\ \addlinespace[.8ex]

\textbf{Positive ganze Zahlen}
\UebFormel{\PositiveIntegers\coloneqq
\{b\in\mathbb Z\mid\IntZero<b\}.}
\UebQuellen{\IntegersResultRef{PositiveIntegersDef}[def]}
&
\textit{Kleinste positive Zahl und Faktorkürzung}
Für \(b\in\PositiveIntegers\) gilt
\UebFormel{\begin{aligned}
&\IntOne\leq b,\\
&z\leq w\leftrightarrow zb\leq wb,\\
&zb=wb\vdash z=w.
\end{aligned}}
\UebQuellen{\IntegersResultRef{PositiveIntegerOneLowerBound}[thm];
\IntegersResultRef{IntegerPositiveFactorOrder}[thm];
\IntegersResultRef{IntegerPositiveFactorCancellation}[thm]}
\\ \addlinespace[.8ex]

\textbf{Zweiseitige Peano-Struktur}
Zur geordneten Arithmetik tritt das Induktionsaxiom mit den Daten
\UebFormel{\begin{aligned}
&P(\IntZero),\\
&\forall z\in\mathbb Z\;(P(z)\to P(z+\IntOne)),\\
&\forall z\in\mathbb Z\;(P(z)\to P(z-\IntOne)).
\end{aligned}}
\UebQuellen{\IntegersResultRef{IntegerPeanoAxioms}[def];
\IntegersResultRef{IntegerTwoSidedInductionAxiom}[ax]}
&
\textit{Modellnachweis und Induktion im Quotienten}
Aus den links genannten Daten folgt im konstruierten Modell
\UebFormel{\forall z\in\mathbb Z\;P(z).}
Der Quotient erfüllt die vollständige geordnete Ganzzahlarithmetik
einschließlich der inversen Schritte und dieser Induktion.
\UebQuellen{\IntegersResultRef{IntegerTwoSidedInductionModel}[thm];
\IntegersResultRef{IntQuotientModelsIntegerPeano}[thm]}
\\
