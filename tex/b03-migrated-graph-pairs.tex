\subsection{Adjunktion eines geordneten Paars}

\Needspace{16\baselineskip}
\FormulaThmDeltaK[Paarzugehörigkeit nach einer Adjunktion]{%
  (i,b)\in R\cup\{(n,a)\}\eqvdash
    (i,b)\in R\lor(i=n\land b=a)
}{B27AdjoinedPairMembership}{\DeltaRow{Mengen}{R\dsep i\dsep b\dsep n\dsep a}}
\begin{tabproofwide}
  \proofstepwidestar[]{(i,b)\in R\cup\{(n,a)\}\leftrightarrow
    ((i,b)\in R\lor(i,b)\in\{(n,a)\})}{%
    \FormulaRefAuto{x\in A\cup B\eqvdash x\in A\lor x\in B}[thm]}
  \proofstepwidestar[]{(i,b)\in\{(n,a)\}\leftrightarrow(i,b)=(n,a)}{%
    \FormulaRefAuto{x\in\{a\}\eqvdash x=a}[thm]}
  \proofstepwidestar[]{(i,b)=(n,a)\leftrightarrow(i=n\land b=a)}{%
    \FormulaRefAuto{(a,b)=(c,d)\eqvdash a=c\land b=d}[thm]}
  \proofstepwidestar[]{\begin{aligned}[t]
    &(i,b)\in R\cup\{(n,a)\}\leftrightarrow{}\\[-2pt]
    &\quad((i,b)\in R\lor(i=n\land b=a))
  \end{aligned}}{\rLRS{3,\rLRS{2,1}}}
\end{tabproofwide}
