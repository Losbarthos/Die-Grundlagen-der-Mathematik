\FormulaThmDeltaK[Ein Paar mit gleichwertiger Zugehörigkeit]{%
  u\in P\leftrightarrow v\in P\vdash
  \{u,v\}\cap P=\varnothing\lor\{u,v\}\cap P=\{u,v\}%
}{B27PairIntersectionFullness}{%
  \DeltaRow{Mengen}{u\dsep v\dsep P\dsep x}
}
\begin{tabproofwide}
  \proofstepwidestar[1]{u\in P\leftrightarrow v\in P}{\rA}
  \proofstepwidestar[]{u\in P\lor u\notin P}{\FormulaRefAuto{P\lor\neg P}[thm]}
  \proofstepwidestar[3]{u\in P}{\rA}
  \proofstepwidestar[1,3]{v\in P}{\rRE{\rLREa{1},3}}
  \proofstepwidestar[5]{x\in\{u,v\}}{\rA}
  \proofstepwidestar[5]{x=u\lor x=v}{\FormulaRefAuto{x\in\{a,b\}\eqvdash(x=a\lor x=b)}[thm]{5}}
  \proofstepwidestar[7]{x=u}{\rA}
  \proofstepwidestar[3,7]{x\in P}{\rIE{\FormulaRefAuto{a=b\vdash b=a}[thm]{7},3}}
  \proofstepwidestar[9]{x=v}{\rA}
  \proofstepwidestar[1,3,9]{x\in P}{\rIE{\FormulaRefAuto{a=b\vdash b=a}[thm]{9},4}}
  \proofstepwidestar[1,3,5]{x\in P}{\rOE{6,7,8,9,10}}
  \proofstepwidestar[1,3]{\{u,v\}\subseteq P}{%
    \FormulaRefAuto{SubsetIntroductionRule}[thm]{[5]\vdots11}}
  \proofstepwidestar[1,3]{\{u,v\}\cap P=\{u,v\}}{%
    \FormulaRefAuto{A\subseteq B\eqvdash A\cap B=A}[thm]{12}}
  \proofstepwidestar[14]{u\notin P}{\rA}
  \proofstepwidestar[1,14]{v\notin P}{%
    \FormulaRefAuto{P\rightarrow Q,\neg Q\vdash\neg P}[thm]{\rLREb{1},14}}
  \proofstepwidestar[1,14]{P\cap\{u,v\}=\varnothing}{%
    \FormulaRefAuto{PairSetDisjointFromTwoAvoided}[thm]{14,15}}
  \proofstepwidestar[1,14]{\{u,v\}\cap P=\varnothing}{%
    \rIE{\FormulaRefAuto{A\cap B=B\cap A}[thm],16}}
  \proofstepwidestar[1]{\{u,v\}\cap P=\varnothing\lor\{u,v\}\cap P=\{u,v\}}{%
    \rOE{2,3,\rOIb{13},14,\rOIa{17}}}
\end{tabproofwide}

\FormulaThmDeltaK[Eine leere Menge oder ein verschiedenes Paar]{%
  C=\varnothing\lor C=\{u,v\}\dsep u\neq v\vdash
  C=\varnothing\lor\exists a\in C\,\exists b\in C\,(a\neq b\land C=\{a,b\})%
}{B27EmptyOrDistinctPairWitnesses}{%
  \DeltaRow{Mengen}{C\dsep u\dsep v\dsep a\dsep b}
}
\begin{tabproofwide}
  \proofstepwidestar[1]{C=\varnothing\lor C=\{u,v\}}{\rA}
  \proofstepwidestar[2]{u\neq v}{\rA}
  \proofstepwidestar[3]{C=\varnothing}{\rA}
  \proofstepwidestar[3]{C=\varnothing\lor\exists a\in C\,\exists b\in C\,(a\neq b\land C=\{a,b\})}{\rOIa{3}}
  \proofstepwidestar[5]{C=\{u,v\}}{\rA}
  \proofstepwidestar[5]{u\in C}{\rIE{\FormulaRefAuto{a=b\vdash b=a}[thm]{5},\FormulaRefAuto{a\in\{a,b\}}[thm]}}
  \proofstepwidestar[5]{v\in C}{\rIE{\FormulaRefAuto{a=b\vdash b=a}[thm]{5},\FormulaRefAuto{b\in\{a,b\}}[thm]}}
  \proofstepwidestar[2,5]{\exists a\in C\,\exists b\in C\,(a\neq b\land C=\{a,b\})}{%
    \rEI{\rAI{6,\rEI{\rAI{7,\rAI{2,5}}}}}}
  \proofstepwidestar[2,5]{C=\varnothing\lor\exists a\in C\,\exists b\in C\,(a\neq b\land C=\{a,b\})}{\rOIb{8}}
  \proofstepwidestar[1,2]{C=\varnothing\lor\exists a\in C\,\exists b\in C\,(a\neq b\land C=\{a,b\})}{\rOE{1,3,4,5,9}}
\end{tabproofwide}
