Die durch einen Mengenterm beschriebenen Bilder erhalten nun eine eigene
Schreibweise. Dabei ist \(t(j)\) ein Mengenterm; die angezeigten
Bindungsvariablen dürfen nicht in seinen festen Parametern vorkommen.

\FormulaDefDeltaK[Durch einen Term gebildete Menge]{%
  \{t(j)\mid j\in J\}\coloneqq
  \iota M\,\forall z\,(z\in M\leftrightarrow
    \exists j\in J\;z=t(j))
}{B27TermImageDef}{\DeltaRow{Mengen}{J\dsep M\dsep j\dsep z}
  \DeltaRow{Mengenterm}{t(j)}}
\begin{tabproofwide}
  \proofstepwidestar[]{\exists!M\,\forall z\,(z\in M\leftrightarrow
    \exists j\in J\;z=t(j))}{\FormulaRefAuto{TermImageSetUniqueExists}[thm]}
\end{tabproofwide}

\FormulaThmDeltaK[Termbildung über einer Vereinigung]{%
  \{t(j)\mid j\in J\cup K\}
    =\{t(j)\mid j\in J\}\cup\{t(j)\mid j\in K\}
}{B27TermImageUnion}{\DeltaRow{Mengen}{J\dsep K\dsep j\dsep z}
  \DeltaRow{Mengenterm}{t(j)}}
\begin{tabproofwide}
  \proofstepwidestar[]{\begin{aligned}[t]
    &z\in\{t(j)\mid j\in J\cup K\}\\[-2pt]
    &\quad\leftrightarrow\exists j\,(j\in J\cup K\land z=t(j))
  \end{aligned}}{%
    \FormulaRefAuto{B27TermImageMembership}[thm]}
  \proofstepwidestar[]{j\in J\cup K\leftrightarrow(j\in J\lor j\in K)}{%
    \FormulaRefAuto{x\in A\cup B\eqvdash x\in A\lor x\in B}[thm]}
  \proofstepwidestar[]{\begin{aligned}[t]
    &(j\in J\cup K\land z=t(j))\leftrightarrow{}\\[-2pt]
    &\quad((j\in J\land z=t(j))\lor(j\in K\land z=t(j)))
  \end{aligned}}{\rLRS{2,\FormulaRefAuto{(P\lor Q)\land R\eqvdash(P\land R)\lor(Q\land R)}[thm]}}
  \proofstepwidestar[]{\begin{aligned}[t]
    &z\in\{t(j)\mid j\in J\cup K\}\leftrightarrow{}\\[-2pt]
    &\quad\exists j\,((j\in J\land z=t(j))\lor(j\in K\land z=t(j)))
  \end{aligned}}{\rLRS{3,1}}
  \proofstepwidestar[]{\begin{aligned}[t]
    &\exists j\,((j\in J\land z=t(j))\lor(j\in K\land z=t(j)))\\[-2pt]
    &\quad\leftrightarrow((\exists j\in J\;z=t(j))\lor(\exists j\in K\;z=t(j)))
  \end{aligned}}{\FormulaRefAuto{\exists x(F(x)\lor G(x))\eqvdash\exists x(F(x))\lor\exists x(G(x))}[thm]}
  \proofstepwidestar[]{\begin{aligned}[t]
    &z\in\{t(j)\mid j\in J\cup K\}\leftrightarrow{}\\[-2pt]
    &\quad((\exists j\in J\;z=t(j))\lor(\exists j\in K\;z=t(j)))
  \end{aligned}}{\rLRS{5,4}}
  \proofstepwidestar[]{\begin{aligned}[t]
    &z\in\{t(j)\mid j\in J\}\cup\{t(j)\mid j\in K\}\leftrightarrow{}\\[-2pt]
    &\quad((\exists j\in J\;z=t(j))\lor(\exists j\in K\;z=t(j)))
  \end{aligned}}{\rLRS{\FormulaRefAuto{B27TermImageMembership}[thm],
    \rLRS{\FormulaRefAuto{B27TermImageMembership}[thm],
      \FormulaRefAuto{x\in A\cup B\eqvdash x\in A\lor x\in B}[thm]}}}
  \proofstepwidestar[]{\begin{aligned}[t]
    &z\in\{t(j)\mid j\in J\cup K\}\leftrightarrow{}\\[-2pt]
    &\quad z\in\{t(j)\mid j\in J\}\cup\{t(j)\mid j\in K\}
  \end{aligned}}{\rLRS{7,6}}
  \proofstepwidestar[]{\begin{aligned}[t]
    &\{t(j)\mid j\in J\cup K\}\\[-2pt]
    &\quad=\{t(j)\mid j\in J\}\cup\{t(j)\mid j\in K\}
  \end{aligned}}{%
    \FormulaRefAuto{\forall x\,(x\in A\leftrightarrow x\in B)\eqvdash A=B}[thm]{\rUI{8}}}
