\providecommand{\FranklRareTripleStatement}{%
  \FormulaThmDeltaK[Eine enthaltene Dreiermenge ohne halbhäufiges Element]{%
    \begin{aligned}[t]
      \exists M\,\exists a\,\exists b\,\exists c\,\bigl(&
        \Finite M\land\UnionClosedFam(M)\land{}\\[-2pt]
        &a\neq b\land a\neq c\land b\neq c\land{}\\[-2pt]
        &\{a,b,c\}\in M\land{}\\[-2pt]
        &\forall x\in\{a,b,c\}\,\neg\HalfOcc{x}{M}\bigr)
    \end{aligned}
  }{FranklRareTripleExists}{%
    \DeltaRow{Mengen}{M\dsep a\dsep b\dsep c\dsep x}
  }%
}
