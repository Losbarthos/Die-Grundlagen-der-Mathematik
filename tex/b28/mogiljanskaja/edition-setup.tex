  end
  tex.sprint("\\externaldocument{" .. directory .. b28_import .. "}[../_B28.pdf]")
  if tex.jobname == "_B28-mog-reading" then
    tex.sprint("\\externaldocument{" .. directory .. "proofs-reading}[_B28-mog-proofs.pdf]")
  end
  for _, predecessor in ipairs(banddeps.graph.B28.predecessors) do
    local base = banddeps.graph[predecessor].artifact_base
    if predecessor == "B21" then base = directory .. "b21-external" end
    thmlookup.load_registry_file(base .. ".registry.tsv")
  end
  thmlookup.load_registry_file(directory .. b28_import .. ".registry.tsv")
  if tex.jobname == "_B28-mog-reading" then
    thmlookup.load_registry_file(directory .. "proofs-reading.registry.tsv")
  end
