% Verweise zwischen Fachband, Lesefassung und Beweistabellen.
\begin{luacode*}
  local job = tex.jobname or ""
  local directory, main_directory = "registry/bracketing/", "registry/"
  if job:match("^_B[0-9][0-9]$") then
    directory, main_directory = "bracketing/", ""
  end
  if job:match("^_B28%-bracketing%-") then
    directory, main_directory = "", "../"
  end
  tex.sprint("\\gdef\\BracketingEditionDirectory{" .. directory .. "}")
  tex.sprint("\\gdef\\BracketingMainDirectory{" .. main_directory .. "}")
\end{luacode*}
