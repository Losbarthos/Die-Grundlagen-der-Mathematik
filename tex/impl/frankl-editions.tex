% Verweise zwischen Band 46 und den Ergänzungen zu Frankls Spezialfällen.
\begin{luacode*}
  local job = tex.jobname or ""
  local directory, main_directory = "registry/frankl/", "registry/"
  if job:match("^_B[0-9][0-9]$") then
    directory, main_directory = "frankl/", ""
  end
  if job:match("^_B46%-frankl%-") then
    directory, main_directory = "", "../"
  end
  tex.sprint("\\gdef\\FranklEditionDirectory{" .. directory .. "}")
  tex.sprint("\\gdef\\FranklMainDirectory{" .. main_directory .. "}")
\end{luacode*}
