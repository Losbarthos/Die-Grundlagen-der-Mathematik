% Navigation zwischen Band 37 und dem Paar über Index und Periode.
\begin{luacode*}
  local job = tex.jobname or ""
  local directory, main_directory = "registry/index-period/", "registry/"
  if job:match("^_B[0-9][0-9]$") then
    directory, main_directory = "index-period/", ""
  elseif job:match("^_B37%-index%-period%-") then
    directory, main_directory = "", "../"
  elseif job:match("^_B[0-9][0-9]%-.+") then
    directory, main_directory = "../index-period/", "../"
  end
  tex.sprint("\\gdef\\IndexPeriodEditionDirectory{" .. directory .. "}")
  tex.sprint("\\gdef\\IndexPeriodMainDirectory{" .. main_directory .. "}")
\end{luacode*}
