% Navigation zwischen Band 28 und den Potenzhalbgruppen-Ergänzungen.
\begin{luacode*}
  local job = tex.jobname or ""
  local directory, main_directory = "registry/power/", "registry/"
  if job:match("^_B[0-9][0-9]$") then
    directory, main_directory = "power/", ""
  end
  if job:match("^_B28%-power%-") then
    directory, main_directory = "", "../"
  end
  tex.sprint("\\gdef\\PowerEditionDirectory{" .. directory .. "}")
  tex.sprint("\\gdef\\PowerMainDirectory{" .. main_directory .. "}")
\end{luacode*}
