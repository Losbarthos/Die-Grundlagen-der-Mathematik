\newcommand{\ProofReferenceMarkStart}[2]{%
  \leavevmode
  \begingroup
  \edef\ProofReferenceMarkNext{\endgroup
    \noexpand\latelua{proofreferences.begin_cell("#1",\number\dimexpr#2\relax,
      \number\ProofReferencesPageTop,\number\ProofReferencesPageBottom,

\newcommand{\ProofReferenceMarkEnd}{\latelua{proofreferences.end_cell()}}
% Reflow the already typeset mathematical nodes: no formula token or premise is
% replaced. Each resulting line becomes a separate, page-breakable table row.
\NewDocumentCommand{\ProofReferenceMathRows}{+m +m}{%
  \ProofReferenceFlowWidth=\dimexpr\linewidth-2\ProofReferencePadding\relax
  % Do not add LaTeX's artificial color scope around the entire formula: real
  % local graphic/color scopes remain in the nodes and are kept together.
  \setbox\ProofReferenceBox=\hbox{$\displaystyle #1$}%
  \directlua{proofreferenceflow.prepare(\number\ProofReferenceBox,
    \number\ProofReferenceFlowWidth,\number\ProofReferenceMaxHeight,
    "\luaescapestring{\detokenize{#2}}")}%
}
\NewDocumentCommand{\ProofReferenceParagraphRows}{+m +m}{%
  \ProofReferenceFlowWidth=\dimexpr\linewidth-2\ProofReferencePadding\relax
  \setbox\ProofReferenceBox=\vbox{%
    \hsize=\ProofReferenceFlowWidth\linewidth=\hsize
    \parindent=0pt\leftskip=0pt\rightskip=0pt plus 1fil
    \parfillskip=0pt\emergencystretch=1em
    \noindent #1\par
  }%
  \directlua{proofreferenceflow.prepare_text(\number\ProofReferenceBox,
    \number\ProofReferenceFlowWidth,\number\ProofReferenceMaxHeight,
    "\luaescapestring{\detokenize{#2}}")}%
}
\NewDocumentCommand{\ProofReferenceFlowLine}{m +m}{%
  \directlua{proofreferenceflow.take(#1,\number\ProofReferenceFlowOutputBox)}%
  \ProofReferenceFlowWidth=\dimexpr\linewidth-2\ProofReferencePadding\relax
  % A genuinely indivisible atom (for example one fraction) is the only
  % fallback case. Bound both dimensions, including the depth below baseline.
  \ifdim\wd\ProofReferenceFlowOutputBox>\ProofReferenceFlowWidth
    \sbox{\ProofReferenceFlowOutputBox}{%
      \resizebox{\ProofReferenceFlowWidth}{!}{\usebox{\ProofReferenceFlowOutputBox}}}%
  \fi
  \ifdim\dimexpr\ht\ProofReferenceFlowOutputBox+\dp\ProofReferenceFlowOutputBox\relax>\ProofReferenceMaxHeight
    \sbox{\ProofReferenceFlowOutputBox}{%
      \resizebox*{!}{\ProofReferenceMaxHeight}{\usebox{\ProofReferenceFlowOutputBox}}}%
  \fi
  \leavevmode\ProofReferenceMarkStart{ProofReferenceCell}{\linewidth}%
  \begin{minipage}[t]{\linewidth}%
    \noindent\hspace*{\ProofReferencePadding}\usebox{\ProofReferenceFlowOutputBox}%
    \par\kern.6ex
  \end{minipage}%
  \ProofReferenceMarkEnd & #2\\[.6ex]
}
\newcommand{\ProofReferenceLink}[3]{%
  \sbox{\ProofReferenceBox}{\ThmLookupEmitRefResolved{#1}{#2}{#3}}%
  \ProofReferenceMarkStart{ProofReferenceCell}{\linewidth}%
  \ifdim\wd\ProofReferenceBox>\linewidth
    \resizebox{\linewidth}{!}{\usebox{\ProofReferenceBox}}%
  \else
    \usebox{\ProofReferenceBox}%
  \fi
  \ProofReferenceMarkEnd
}

\newcommand{\ProofReferenceHeaderRule}{%
  \noalign{\hbox{%
    \ProofReferenceMarkStart{ProofReferenceHeader}{\ProofReferencesTableWidth}%
    \vrule width\ProofReferencesTableWidth height\arrayrulewidth depth0pt
    \ProofReferenceMarkEnd}}%
}
\NewDocumentCommand{\ProofReferencesTable}{+m}{%
  \par\begingroup
  \ProofReferencesRecordingfalse
  \normalfont\small
  \ProofReferencesTableWidth=\linewidth
  \ProofReferencePadding=.4em
  \ProofReferenceMaxHeight=.65\textheight
  \ProofReferencePageReserve=.5\baselineskip
  \ProofReferencesPageTop=\dimexpr1in+\voffset+\topmargin+\headheight+\headsep\relax
  \ProofReferencesPageBottom=\dimexpr\ProofReferencesPageTop+\textheight\relax
  \ProofReferencesOddLeft=\dimexpr1in+\hoffset+\oddsidemargin\relax
  \ProofReferencesEvenLeft=\ProofReferencesOddLeft
  \csname if@twoside\endcsname
    \ProofReferencesEvenLeft=\dimexpr1in+\hoffset+\evensidemargin\relax
  \fi
  \setlength{\LTleft}{0pt}\setlength{\LTright}{0pt}%
  \setlength{\LTpre}{.5\baselineskip}%
  \renewcommand{\arraystretch}{1.15}%
  \begin{longtable}{@{}p{.77\linewidth}@{\hspace{.02\linewidth}}%
    >{\raggedright\arraybackslash}p{.21\linewidth}@{}}
    \multicolumn{2}{@{}p{\ProofReferencesTableWidth}@{}}{%
      \ProofReferenceMarkStart{ProofReferenceHeader}{\ProofReferencesTableWidth}%
      \textbf{\ProofReferencesTitle}\ProofReferenceMarkEnd}\\[.4ex]
    \ProofReferenceMarkStart{ProofReferenceHeader}{\linewidth}%
    \textbf{Aussage mit Voraussetzungen}\ProofReferenceMarkEnd &
    \ProofReferenceMarkStart{ProofReferenceHeader}{\linewidth}%
    \textbf{Verweis}\ProofReferenceMarkEnd\\\ProofReferenceHeaderRule
    \endfirsthead
    \multicolumn{2}{@{}p{\ProofReferencesTableWidth}@{}}{%
      \ProofReferenceMarkStart{ProofReferenceHeader}{\ProofReferencesTableWidth}%
      \textbf{\ProofReferencesTitle{} (Fortsetzung)}\ProofReferenceMarkEnd}\\[.4ex]
    \ProofReferenceMarkStart{ProofReferenceHeader}{\linewidth}%
    \textbf{Aussage mit Voraussetzungen}\ProofReferenceMarkEnd &
    \ProofReferenceMarkStart{ProofReferenceHeader}{\linewidth}%
    \textbf{Verweis}\ProofReferenceMarkEnd\\\ProofReferenceHeaderRule
    \endhead
    % Font bounding boxes can extend beyond TeX's visible glyph depth. Keep
    % a small reserve on every page, including the final table page.
    \noalign{\kern\ProofReferencePageReserve}
    \endfoot
    #1%
    % Flush the depth of a tall final row while longtable's output routine is
    % still active; otherwise it can move to a new page without its header.
    \noalign{\penalty0\vskip0pt}%
  \end{longtable}
  \endgroup
}

    \expandafter\ProofReferencesCaptureBody\expandafter

  \fi
}

}
\NewDocumentCommand{\ProofReferenceText}{+m}{%
  \vcenter{\hbox{\begin{minipage}{\ProofReferenceFlowWidth}%
    \normalfont\small
    \let\label\@gobble
    \RenewDocumentCommand{\ThmLookupRegister}{m m m m g}{}%
    \RenewDocumentCommand{\ThmLookupRegisterId}{m m m}{}%

    #1%
  \end{minipage}}}%
}
