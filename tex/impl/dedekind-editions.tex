% Verweise auf die eigenständigen Ergänzungen zum Rekursionssatz.
\begin{luacode*}
  local job = tex.jobname or ""
  local directory = "registry/dedekind/"
  if job:match("^_B[0-9][0-9]$") then directory = "dedekind/" end
  if job:match("^_B10%-dedekind%-") then directory = "" end
  tex.sprint("\\gdef\\DedekindEditionDirectory{" .. directory .. "}")
\end{luacode*}

  \href{\DedekindEditionDirectory_B10-dedekind-reading.pdf\#dedekind.reading}{Lesefassung}%
}
