% Navigation zwischen Band 37 und den Nullprodukt-Ergänzungen.
\begin{luacode*}
  local job = tex.jobname or ""
  local directory, main_directory = "registry/nullproducts/", "registry/"
  if job:match("^_B[0-9][0-9]$") then
    directory, main_directory = "nullproducts/", ""
  end
  if job:match("^_B37%-nullproducts%-") then
    directory, main_directory = "", "../"
  end
  tex.sprint("\\gdef\\NullproductsEditionDirectory{" .. directory .. "}")
  tex.sprint("\\gdef\\NullproductsMainDirectory{" .. main_directory .. "}")
\end{luacode*}
