% Verweise zwischen Band 40 und den Ergänzungen zur Gruppenrekonstruktion.
\begin{luacode*}
  local job = tex.jobname or ""
  local directory, main_directory = "registry/reconstruction/", "registry/"
  if job:match("^_B[0-9][0-9]$") then
    directory, main_directory = "reconstruction/", ""
  end
  if job:match("^_B40%-reconstruction%-") then
    directory, main_directory = "", "../"
  end
  tex.sprint("\\gdef\\ReconstructionEditionDirectory{" .. directory .. "}")
  tex.sprint("\\gdef\\ReconstructionMainDirectory{" .. main_directory .. "}")
\end{luacode*}
