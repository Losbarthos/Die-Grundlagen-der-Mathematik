% Navigation zwischen Hauptband, Lesefassung und Beweistabellen.
\begin{luacode*}
  local job = tex.jobname or ""
  local directory, main_directory = "registry/power-isomorphisms/", "registry/"
  if job:match("^_B[0-9][0-9]$") then
    directory, main_directory = "power-isomorphisms/", ""
  end
  if job:match("^_B28%-power%-isomorphisms%-") then
    directory, main_directory = "", "../"
  end
  tex.sprint("\\gdef\\PowerIsoEditionDirectory{" .. directory .. "}")
  tex.sprint("\\gdef\\PowerIsoMainDirectory{" .. main_directory .. "}")
  if job == "_B28" or job == "B28" or job == "main" or job == "Bd. 28 - Halbgruppen" then
    tex.sprint("\\externaldocument{registry/power-isomorphisms/proofs-main}[" .. directory .. "_B28-power-isomorphisms-proofs.pdf]")
  end
\end{luacode*}
