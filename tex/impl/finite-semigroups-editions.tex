% Navigation zur Lesefassung des Gesamtwegs von Band 37.
\begin{luacode*}
  local job = tex.jobname or ""
  local directory, main_directory, null_directory = "registry/finite/", "registry/", "registry/nullproducts/"
  if job:match("^_B[0-9][0-9]$") then
    directory, main_directory, null_directory = "finite/", "", "nullproducts/"
  end
  if job == "_B37-finite-reading" then
    directory, main_directory, null_directory = "", "../", "../nullproducts/"
  end
  if job:match("^_B37%-nullproducts%-") then
    directory, main_directory, null_directory = "../finite/", "../", ""
  end
  tex.sprint("\\gdef\\FiniteSemigroupsEditionDirectory{" .. directory .. "}")
  tex.sprint("\\gdef\\FiniteSemigroupsMainDirectory{" .. main_directory .. "}")
  tex.sprint("\\gdef\\FiniteSemigroupsNullproductsDirectory{" .. null_directory .. "}")
\end{luacode*}
