% Verweise zwischen Band 45 und den Ergänzungen zu Halbverbänden und Ordnung.
\begin{luacode*}
  local job = tex.jobname or ""
  local directory, main_directory = "registry/semilattice/", "registry/"
  if job:match("^_B[0-9][0-9]$") then
    directory, main_directory = "semilattice/", ""
  end
  if job:match("^_B45%-semilattice%-") then
    directory, main_directory = "", "../"
  end
  tex.sprint("\\gdef\\SemilatticeEditionDirectory{" .. directory .. "}")
  tex.sprint("\\gdef\\SemilatticeMainDirectory{" .. main_directory .. "}")
\end{luacode*}
