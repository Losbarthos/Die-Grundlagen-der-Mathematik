% tex/impl/commands/words-trees.tex
% Wiederverwendbare Notation für endliche Wörter und Klammerungsbäume.

\ProvidesFile{words-trees.tex}[2026/08/30 Word and bracketing-tree notation]

\DeclareRobustCommand{\FiniteWords}[1]{%
  \ensuremath{{#1}^{\ast}}%
}
\DeclareRobustCommand{\NonemptyWords}[1]{%
  \ensuremath{{#1}^{+}}%
}
\DeclareRobustCommand{\WordLength}[1]{%
  \ensuremath{\lvert #1\rvert}%
}
\DeclareRobustCommand{\EmptyWord}[1]{%
  \ensuremath{\varepsilon_{#1}}%
}
\DeclareRobustCommand{\LetterWord}[1]{%
  \ensuremath{\langle #1\rangle}%
}
\DeclareRobustCommand{\WordConcat}[2]{%
  \ensuremath{(#1\mathbin{\frown}#2)}%
}
\DeclareRobustCommand{\WordFold}[2]{%
  \ensuremath{\Pi_{#1}\!\left(#2\right)}%
}
\DeclareRobustCommand{\BinaryOpFunction}[2]{%
  \ensuremath{\widehat{#2}_{#1}}%
}
\DeclareRobustCommand{\WordRumpfMap}[1]{%
  \ensuremath{\operatorname{rumpf}_{#1}}%
}
\DeclareRobustCommand{\WordRumpf}[2]{%
  \ensuremath{\WordRumpfMap{#1}\!\left(#2\right)}%
}
\DeclareRobustCommand{\WordEndMap}[1]{%
  \ensuremath{\operatorname{ende}_{#1}}%
}
\DeclareRobustCommand{\WordEnd}[2]{%
  \ensuremath{\WordEndMap{#1}\!\left(#2\right)}%
}
\DeclareRobustCommand{\TreeRecursor}[4]{%
  \ensuremath{\operatorname{Rek}_{#1,#2}\!\left(#3,#4\right)}%
}
\DeclareRobustCommand{\LeftBracketingMap}[1]{%
  \ensuremath{L_{#1}}%
}
\DeclareRobustCommand{\LeftBracketing}[2]{%
  \ensuremath{\LeftBracketingMap{#1}\!\left(#2\right)}%
}

\DeclareRobustCommand{\TreeAlphabet}[1]{%
  \ensuremath{\Sigma_{#1}}%
}
\DeclareRobustCommand{\LeafTree}[2]{%
  \ensuremath{\operatorname{Blatt}_{#1}\!\left(#2\right)}%
}
\DeclareRobustCommand{\NodeTree}[3]{%
  \ensuremath{\operatorname{Knoten}_{#1}\!\left(#2,#3\right)}%
}
\DeclareRobustCommand{\TreeClosure}[2]{%
  \ensuremath{\mathcal C_{#1}\!\left(#2\right)}%
}
\DeclareRobustCommand{\TreeStage}[2]{%
  \ensuremath{\mathcal T_{#1,#2}}%
}
\DeclareRobustCommand{\BracketTreeSet}[1]{%
  \ensuremath{\mathcal T\!\left(#1\right)}%
}
\DeclareRobustCommand{\TreeLeafWord}[2]{%
  \ensuremath{\operatorname{wort}_{#1}\!\left(#2\right)}%
}
\DeclareRobustCommand{\TreeBrackets}[3]{%
  \ensuremath{#2\mathrel{\triangleright_{#1}}#3}%
}
\DeclareRobustCommand{\TreeAppendLeaf}[3]{%
  \ensuremath{(#2\mathbin{\triangleleft_{#1}}#3)}%
}
\DeclareRobustCommand{\WordBracketings}[2]{%
  \ensuremath{\operatorname{Kl}_{#1}\!\left(#2\right)}%
}
\DeclareRobustCommand{\TreeEvaluation}[2]{%
  \ensuremath{\operatorname{ev}_{#1}\!\left(#2\right)}%
}
