% tex/impl/commands/cardinality.tex
\ProvidesFile{cardinality.tex}[2026/03/25 Cardinality and finiteness notation]

\providecommand{\EqCard}{\mathrel{\approx}}

\providecommand{\CardLeq}{\mathrel{\preccurlyeq_{\!\mathrm c}}}
\providecommand{\CardLt}{\mathrel{\prec_{\!\mathrm c}}}

\DeclareRobustCommand{\EqCardRel}[1]{%
  \ensuremath{\mathsf{EqCard}_{#1}}%
}

\DeclareRobustCommand{\Finite}[1]{%
  \ensuremath{\mathsf{Finite}\!\left(#1\right)}%
}

\DeclareRobustCommand{\Infinite}[1]{%
  \ensuremath{\mathsf{Infinite}\!\left(#1\right)}%
}

\DeclareRobustCommand{\AtMostCountable}[1]{%
  \ensuremath{\mathsf{Count}_{\leq}\!\left(#1\right)}%
}

\DeclareRobustCommand{\CountablyInfinite}[1]{%
  \ensuremath{\mathsf{Count}_{\infty}\!\left(#1\right)}%
}

\DeclareRobustCommand{\Uncountable}[1]{%
  \ensuremath{\mathsf{Uncount}\!\left(#1\right)}%
}

\DeclareRobustCommand{\SingletonEmbedding}[1]{%
  \ensuremath{\sigma_{#1}}%
}

\DeclareRobustCommand{\CBPart}[2]{%
  \ensuremath{\mathsf C_{#1,#2}}%
}

\makeatletter
\DeclareRobustCommand{\tCard}{%
  \@ifnextchar({\tCard@paren}{%
    \@ifnextchar\bgroup{\tCard@brace}{\ensuremath{t_{\mathsf{Card}}}}%
  }%
}
\def\tCard@paren(#1){%
  \ensuremath{t_{\mathsf{Card}}\!\left(#1\right)}%
}
\def\tCard@brace#1{%
  \ensuremath{t_{\mathsf{Card}}\!\left(#1\right)}%
}
\makeatother
