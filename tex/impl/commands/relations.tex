% tex/impl/commands/relations.tex
\ProvidesFile{relations.tex}[2026/01/17 Relation-theoretic notation]

% ------------------------------------------------------------
% Praedikate / Begriffe zu Relationen.
% Bei Aequivalenz- und Ordnungsrelationen steht der Traeger stets vor dem
% in Infixnotation verwendeten Relationsoperator, also etwa
% \EqRel{A,\sim} beziehungsweise \PartOrd{A,\leq}.
% ------------------------------------------------------------

% Totale Relation: TR(R,A,B)
\providecommand{\TotRel}[1]{\ensuremath{\mathsf{TR}(#1)}}

% Partielle Ordnung: PartOrd(A,\leq)
\providecommand{\PartOrd}[1]{%
  \ensuremath{\mathsf{PartOrd}\!\left(#1\right)}%
}

% Totale Ordnung: TotOrd(A,\leq)
\providecommand{\TotOrd}[1]{%
  \ensuremath{\mathsf{TotOrd}\!\left(#1\right)}%
}

% Aequivalenzrelation: EqRel(A,\sim)
\providecommand{\EqRel}[1]{%
  \ensuremath{\mathsf{EqRel}\!\left(#1\right)}%
}

\DeclareRobustCommand{\EqClass}[3]{%
  \ensuremath{\left[#1\right]_{#2,#3}}%
}

\DeclareRobustCommand{\QuotientSet}[2]{%
  \ensuremath{#1/#2}%
}

\DeclareRobustCommand{\QuotProj}[2]{%
  \ensuremath{q_{#1,#2}}%
}
\DeclareRobustCommand{\tQuotProj}[2]{%
  \ensuremath{t_{\QuotProj{#1}{#2}}}%
}

\DeclareRobustCommand{\StrictSupRel}[1]{R^{\subset}_{#1}}
\DeclareRobustCommand{\FreshRel}[1]{Q_{#1}}

% ------------------------------------------------------------
% Minimum und Maximum einer Teilmenge in einer dargestellten Ordnung.
% ------------------------------------------------------------
\providecommand{\Min}[1]{%
  \ensuremath{\mathop{\min}\nolimits_{#1}}%
}
\providecommand{\Max}[1]{%
  \ensuremath{\mathop{\max}\nolimits_{#1}}%
}

\DeclareMathOperator{\LB}{LB}
\DeclareMathOperator{\UB}{UB}


% ------------------------------------------------------------
% Ordnungs-Prädikate
% ------------------------------------------------------------
\DeclareRobustCommand{\WellOrd}[1]{%
  \ensuremath{\mathrm{WellOrd}\!\bigl(#1\bigr)}%
}
\DeclareRobustCommand{\WellOrdable}[1]{%
  \ensuremath{\mathsf{WellOrdable}\!\bigl(#1\bigr)}%
}
\DeclareRobustCommand{\WOP}{\ensuremath{\mathsf{WO}}}
