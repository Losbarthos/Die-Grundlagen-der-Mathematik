% Navigation zwischen Band 37 und dem Paar über Multiplikationstafeln.
\begin{luacode*}
  local job = tex.jobname or ""
  local directory, main_directory = "registry/table-isomorphisms/", "registry/"
  if job:match("^_B[0-9][0-9]$") then
    directory, main_directory = "table-isomorphisms/", ""
  elseif job:match("^_B37%-table%-isomorphisms%-") then
    directory, main_directory = "", "../"
  elseif job:match("^_B[0-9][0-9]%-.+") then
    directory, main_directory = "../table-isomorphisms/", "../"
  end
  tex.sprint("\\gdef\\TableIsomorphismsEditionDirectory{" .. directory .. "}")
  tex.sprint("\\gdef\\TableIsomorphismsMainDirectory{" .. main_directory .. "}")
  if job == "_B37" or job == "B37" or job == "main" or job == "Bd. 37 - Endliche Halbgruppen" then
    tex.sprint("\\externaldocument{registry/table-isomorphisms/proofs-main}[" .. directory .. "_B37-table-isomorphisms-proofs.pdf]")
  end
\end{luacode*}
