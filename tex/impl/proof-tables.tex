  \def\mxProofPartParent{#1}%

\renewcommand{\theproofpartnr}{\mxProofPartParent(\roman{proofpartnr})}

% interner Helfer: Teil registrieren + Header drucken
% Signatur wie \FormulaThmAuto: [Titel]{Anzeigeformel}[Keyformel][ID]
\NewDocumentCommand{\mxProofPartOpen}{m o m o o}{%
  % #1 = Tabellenpräambel, #2 = optional Titel, #3 = Formel (Anzeige), #4 = optional Key, #5 = optional ID

  \par\medskip
  \refstepcounter{proofpartnr}%

  % Label (stabil/ASCII; ohne Klammern/roman im Labelnamen)
  \edef\mxPPLbl{thm:pp:\mxProofPartParent:\arabic{proofpartnr}}%
  \label{\mxPPLbl}%

  % Key bestimmen (Default: Anzeigeformel)
  \def\mxKey{#3}%
  \IfNoValueF{#4}{%
    \IfBlankTF{#4}{}{%
      \def\mxKey{#4}%
    }%
  }%

  % Registrierung:
  % - Wenn ID angegeben: nur ID->Label registrieren (keine Strukturregistrierung => keine Duplikat-Key-Probleme)
  % - Sonst: Struktur registrieren (wie bei FormulaThmAuto), Referenz dann über \FormulaRefAuto{<Keyformel/Formel>}
  \IfNoValueTF{#5}{%
    \IfNoValueTF{#2}
      {\ThmLookupRegister{\mxPPLbl}{theorem}{\theproofpartnr}{\mxKey}}
      {\ThmLookupRegister{\mxPPLbl}{theorem}{\theproofpartnr}{\mxKey}{#2}}%
  }{%
    \IfBlankTF{#5}{}{%
      \ThmLookupRegisterId{#5}{theorem}{\mxPPLbl}%
    }%
  }%

  % Header: optional Titel links, eigene Referenz rechts
  \noindent
  \IfNoValueTF{#2}{\textbf{Teil}}{\textbf{#2}}%
  \hfill \ThmLookupEmitRef{theorem}{\mxPPLbl}%
  \par
  \[
    #3
  \]
  \par\small

  % Tabellen-Setup
  \setcounter{proofstepnr}{0}%
  \setlength{\tabcolsep}{2pt}%
  \renewcommand{\arraystretch}{1.15}%
  \begin{longtable}{#1}%
}

\newcommand{\mxProofPartClose}{%
  \end{longtable}\par
}

% ------------------------------------------------------------
% ProofParts mit Delta-Kontext (analog zu \FormulaThmDeltaR)
% ------------------------------------------------------------
% Signatur:
% \mxProofPartOpenDelta{<Tabellenpräambel>}[<Titel>]{<Anzeigeformel>}{<DeltaRows>}[<Keyformel>][<ID>]
\NewDocumentCommand{\mxProofPartOpenDelta}{m o m +G{} o o}{%
  % #1 = Tabellenpräambel
  % #2 = optional Titel
  % #3 = Anzeigeformel
  % #4 = DeltaRows (optional, default leer)
  % #5 = optional Keyformel
  % #6 = optional ID

  \par\medskip
  \refstepcounter{proofpartnr}%

  \edef\mxPPLbl{thm:pp:\mxProofPartParent:\arabic{proofpartnr}}%
  \label{\mxPPLbl}%

  % Key bestimmen (Default: Anzeigeformel)
  \def\mxKey{#3}%
  \IfNoValueF{#5}{%
    \IfBlankTF{#5}{}{%
      \def\mxKey{#5}%
    }%
  }%

  % Registrierung (wie bei \mxProofPartOpen)
  \IfNoValueTF{#6}{%
    \IfNoValueTF{#2}
      {\ThmLookupRegister{\mxPPLbl}{theorem}{\theproofpartnr}{\mxKey}}
      {\ThmLookupRegister{\mxPPLbl}{theorem}{\theproofpartnr}{\mxKey}{#2}}%
  }{%
    \IfBlankTF{#6}{}{%
      \ThmLookupRegisterId{#6}{theorem}{\mxPPLbl}%
    }%
  }%

  % Header
  \noindent
  \IfNoValueTF{#2}{\textbf{Teil}}{\textbf{#2}}%
  \hfill \ThmLookupEmitRef{theorem}{\mxPPLbl}%
  \par

  % Delta-Block (analog \FormulaThmDeltaR: leerer Titel)
  \IfBlankTF{#4}{}{%
    \begin{DeltaContext}{ }%
      #4%
    \end{DeltaContext}%
  }%

  % Anzeigeformel
  \[
    #3
  \]
  \par\small

  % Tabellen-Setup
  \setcounter{proofstepnr}{0}%
  \setlength{\tabcolsep}{2pt}%
  \renewcommand{\arraystretch}{1.15}%
  \begin{longtable}{#1}%
}

% Breite Formeln und Begründungen
% ----------------------------------
% Die frühere Break-Implementierung legte die Begründung in eine nullbreite
% Überlagerungsbox. Dadurch konnte sie links aus dem Satzspiegel laufen, ohne
% dass TeX eine Overfull-Warnung erzeugte.  Außerdem wurden auch kurze Zeilen
% stets auf zwei Tabellenzeilen verteilt.
%
% Die bestehenden Break-Aufrufe wählen deshalb nun automatisch:
% - kurze Formel/Begründung-Kombinationen bleiben in einer Zeile;
% - lange Kombinationen erhalten eine echte, auf \linewidth begrenzte
%   Fortsetzungszelle.
%
% Die getrennten Obergrenzen für Formel und Begründung sind absichtlich
% konservativ.  So können nicht die breiteste Formel einer Tabellenzeile und
% die breiteste Begründung einer anderen Zeile als unabhängige Spaltenmaxima
% die longtable künstlich verbreitern.  Die Summengrenze lässt zugleich Platz
% für Abhängigkeitsliste, Schrittnummer und Spaltenabstände.
\newsavebox{\mxProofWideFormulaMeasureBox}
\newsavebox{\mxProofWideReasonMeasureBox}
\newlength{\mxProofWideCombinedMeasure}

% Standardmäßig bleiben ausgelagerte Begründungen wie bisher am rechten
% Satzspiegel. In dicht gesetzten Beweisfolgen kann lokal auf eine
% formelnahe Fortsetzungszeile umgeschaltet werden. Der Modus wird bewusst
% explizit wieder ausgeschaltet, damit bestehende Bände unverändert bleiben.
\newif\ifmxProofWideReasonsNearFormula
\mxProofWideReasonsNearFormulafalse
\newcommand{\ProofWideReasonsNearFormula}{%
  \mxProofWideReasonsNearFormulatrue}
\newcommand{\ProofWideReasonsAtRight}{%
  \mxProofWideReasonsNearFormulafalse}

\newcommand{\mxProofStepWideSetMeasureBoxes}[2]{%
  % Die Boxen werden global gesetzt, weil eine Alignment-Zelle ihre lokale
  % Gruppe am folgenden & beendet.  Beim Einsetzen beziehungsweise Auflösen
  % wird ihr Inhalt sofort in die jeweilige Tabellenzeile kopiert; die nächste
  % Beweiszeile darf die Register daher gefahrlos wiederverwenden.
  \global\setbox\mxProofWideFormulaMeasureBox=\hbox{\ensuremath{#1}}%
  \global\setbox\mxProofWideReasonMeasureBox=\hbox{\ensuremath{#2}}%
}

\newcommand{\mxProofStepWideInlineRow}[1]{%
  #1 & \theproofstepnr
     & \multicolumn{3}{@{}l@{}}{%
         \usebox{\mxProofWideFormulaMeasureBox}}
     & \ifmxProofWideReasonsNearFormula\hspace*{1em}\fi
       \usebox{\mxProofWideReasonMeasureBox}\\%
}

% #1 = Stern für \\* nach der Formelzeile, #2 = Abhängigkeiten. Formel und
% Begründung liegen bereits in den Messboxen. Beide langen Inhalte
% überspannen ihre jeweiligen Gesamtbereiche; dadurch beeinflussen sie nicht
% die Maxima der einzelnen Formel-/Begründungsspalten.
\NewDocumentCommand{\mxProofStepWideForcedBreakRows}{s m}{%
  #2 & \theproofstepnr
     & \multicolumn{4}{@{}l@{}}{%
         \usebox{\mxProofWideFormulaMeasureBox}}%
  \IfBooleanTF{#1}{\\*[-.2ex]}{\\[-.2ex]}%
  \ifmxProofWideReasonsNearFormula
    % Zwei leere Vorspalten richten die Begründung am Beginn des
    % Formelbereichs aus. Die feste Restbreite verhindert zugleich, dass
    % lange Referenzen den Satzspiegel verbreitern.
    & & \multicolumn{4}{@{}>{\raggedright\arraybackslash}p{.80\linewidth}@{}}{%
      \unhcopy\mxProofWideReasonMeasureBox}\\%
  \else
    \multicolumn{6}{@{}>{\raggedleft\arraybackslash}p{\linewidth}@{}}{%
      % Das Auflösen der Messbox stellt die in ihrem Inhalt vorhandenen
      % mathematischen Umbruchstellen im Absatz wieder her. Nicht trennbare
      % Einzelstücke bleiben echte Boxen und erzeugen bei Überbreite eine
      % reguläre TeX-Warnung.
      \unhcopy\mxProofWideReasonMeasureBox}\\%
  \fi
}

% Automatische Auswahl.  Das Messen geschieht vor dem ersten Alignment-&;
% die eigentliche Zeile wird erst nach den abgeschlossenen Vergleichen
% expandiert.  Damit bleiben die Bedingungen sicher innerhalb einer
% longtable-Zelle.
\NewDocumentCommand{\mxProofStepWideAutoRows}{s m m m}{%
  \mxProofStepWideSetMeasureBoxes{#3}{#4}%
  \setlength{\mxProofWideCombinedMeasure}{%
    \wd\mxProofWideFormulaMeasureBox}%
  \addtolength{\mxProofWideCombinedMeasure}{%
    \wd\mxProofWideReasonMeasureBox}%
  % Der kleine Zwischenraum entspricht dem optischen Mindestabstand, den
  % sonst das dehnbare Material vor der Begründung herstellt.
  \addtolength{\mxProofWideCombinedMeasure}{1em}%
  \def\mxProofStepWideNext{\mxProofStepWideInlineRow}%
  \ifdim\wd\mxProofWideFormulaMeasureBox>.55\linewidth
    \IfBooleanTF{#1}
      {\def\mxProofStepWideNext{\mxProofStepWideForcedBreakRows*}}
      {\def\mxProofStepWideNext{\mxProofStepWideForcedBreakRows}}%
  \else
    \ifdim\wd\mxProofWideReasonMeasureBox>.20\linewidth
      \IfBooleanTF{#1}
        {\def\mxProofStepWideNext{\mxProofStepWideForcedBreakRows*}}
        {\def\mxProofStepWideNext{\mxProofStepWideForcedBreakRows}}%
    \else
      \ifdim\mxProofWideCombinedMeasure>.72\linewidth
        \IfBooleanTF{#1}
          {\def\mxProofStepWideNext{\mxProofStepWideForcedBreakRows*}}
          {\def\mxProofStepWideNext{\mxProofStepWideForcedBreakRows}}%
      \fi
    \fi
  \fi
  \mxProofStepWideNext{#2}%
}

% Kompatible automatische Varianten: Signatur und Schrittzählung bleiben

\NewDocumentCommand{\proofstepwidestarbreak}{O{} m m}{%
  \stepcounter{proofstepnr}%

% Wie \proofstepwidestarbreak, jedoch ohne zulässigen Seitenumbruch zwischen

  \stepcounter{proofstepnr}%
