\begin{luacode*}
  local job = tex.jobname or ""
  local directory, main_directory = "registry/coordinates/", "registry/"
  if job:match("^_B[0-9][0-9]$") then
    directory, main_directory = "coordinates/", ""
  end
  if job:match("^_B31%-coordinates%-") then
    directory, main_directory = "", "../"
  end
  tex.sprint("\\gdef\\CoordinatesEditionDirectory{" .. directory .. "}")
  tex.sprint("\\gdef\\CoordinatesMainDirectory{" .. main_directory .. "}")
\end{luacode*}
