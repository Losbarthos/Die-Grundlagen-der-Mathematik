% Navigation zwischen Band 47 und den Kompaktheits-Ergänzungen.
\begin{luacode*}
  local job = tex.jobname or ""
  local directory, main_directory = "registry/compactness/", "registry/"
  if job:match("^_B[0-9][0-9]$") then
    directory, main_directory = "compactness/", ""
  end
  if job:match("^_B47%-compactness%-") then
    directory, main_directory = "", "../"
  end
  tex.sprint("\\gdef\\CompactnessEditionDirectory{" .. directory .. "}")
  tex.sprint("\\gdef\\CompactnessMainDirectory{" .. main_directory .. "}")
\end{luacode*}
