  local directory = "registry/integers/"
  local reading = tex.jobname == "_B17-integers-reading"
  thmlookup.registry_path = directory .. tex.jobname .. ".registry.tsv"
  thmlookup.debug_path = directory .. tex.jobname .. ".debug.log"
  thmlookup.prepare_run()
  for _, predecessor in ipairs(banddeps.graph.B17.predecessors) do
    local base = banddeps.graph[predecessor].artifact_base
    tex.sprint("\\externaldocument{" .. base .. "}[../_" .. predecessor .. ".pdf]")
  end
  tex.sprint("\\externaldocument{" .. directory .. "b17-external}[../_B17.pdf]")
  if reading then
    tex.sprint("\\externaldocument{" .. directory .. "proofs-reading}[_B17-integers-proofs.pdf]")
  end
  banddeps.load_predecessor_registries("B17")
  thmlookup.load_registry_file(directory .. "b17-external.registry.tsv")
  if reading then thmlookup.load_registry_file(directory .. "proofs-reading.registry.tsv") end
\end{luacode*}
