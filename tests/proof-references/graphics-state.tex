\directlua{
  fixture_graphics_seen = {}
  local watched = {}
  for _, name in ipairs({"pdf_save", "pdf_setmatrix", "pdf_restore", "pdf_colorstack",
                        "pdf_start_link", "pdf_end_link"}) do
    watched[node.subtype(name)] = name
  end
  local function inspect(head)
    for item in node.traverse(head) do
      if item.id == node.id("whatsit") and watched[item.subtype] then
        local name = watched[item.subtype]
        fixture_graphics_seen[name] = (fixture_graphics_seen[name] or 0) + 1
      end
      if item.list then inspect(item.list) end
    end
  end
  local original_prepare = proofreferenceflow.prepare
  proofreferenceflow.prepare = function(boxnum, ...)
    inspect(tex.box[boxnum].list)
    return original_prepare(boxnum, ...)
  end
}
\AtEndDocument{\directlua{
  local output = assert(io.open(fixture_output .. "/graphics-state.graphics.tsv", "w"))
  for _, name in ipairs({"pdf_save", "pdf_setmatrix", "pdf_restore", "pdf_colorstack",
                        "pdf_start_link", "pdf_end_link"}) do
    output:write(name, string.char(9), tostring(fixture_graphics_seen[name] or 0), string.char(10))
  end
  output:close()
