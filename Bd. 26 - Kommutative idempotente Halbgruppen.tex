\chapter{Kommutative idempotente Halbgruppen}

\section{Definition}

\FormulaDefDeltaK[Kommutative idempotente Halbgruppe]{%
  \CommutativeIdempotentSemigroupAlg{A}{\star}%
}{CommutativeIdempotentSemigroupDef}{%
  \DeltaRow{Mengen}{A}
  \DeltaRow{\textbf{Axiome}}{}
  \DeltaPrem{Kommutative Halbgruppe}{%
    \CommutativeSemigroupAlg{A}{\star}}
  \DeltaPrem{Idempotente Halbgruppe}{%
    \IdempotentSemigroupAlg{A}{\star}}
  \DeltaRow{\textbf{Neues Symbol}}{}
  \DeltaPrem{\makecell[l]{Kommutative idempotente\\ Halbgruppe}}{%
    \CommutativeIdempotentSemigroupAlg{A}{\star}}

  \DeltaRow{Mengen}{A}

\FormulaAxiomDeltaK[Zugriff auf die idempotente Halbgruppe]{%

}{CommutativeIdempotentSemigroupIdempotentBaseAxiom}{%
  \DeltaRow{Mengen}{A}

\FormulaAxiomDeltaK[Zugriff auf die Idempotenz der kommutativen Struktur]{%
  \CommutativeIdempotentSemigroupAlg{A}{\star}\dsep x\in A
  \vdash x\star x=x%
}{CommutativeIdempotentSemigroupIdempotenceAxiom}{%
  \DeltaRow{Mengen}{A\dsep x}
}

\FormulaAxiomDeltaK[Zugriff auf die Halbgruppe]{%
  \CommutativeIdempotentSemigroupAlg{A}{\star}
  \vdash\SemigroupAlg{A}{\star}%
}{CommutativeIdempotentSemigroupSemigroupAxiom}{%
  \DeltaRow{Mengen}{A}
}

\FormulaAxiomDeltaK[Zugriff auf die Kommutativität]{%
  \CommutativeIdempotentSemigroupAlg{A}{\star}\dsep x,y\in A
  \vdash x\star y=y\star x%
}{CommutativeIdempotentSemigroupCommutativityAxiom}{%
  \DeltaRow{Mengen}{A\dsep x\dsep y}
}
