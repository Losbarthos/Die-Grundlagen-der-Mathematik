\chapter{Kürzbare Halbgruppen}

\section{Definition der Kürzbarkeit}

\FormulaDefDeltaK[Kürzbare Operation]{%
  \mathsf{K\ddot{u}rz}\!\left(A,\star\right)%
}{CancellativeOperationDef}{%
  \DeltaRow{Mengen}{A}
  \DeltaPrem{Linkskürzbarkeit}{\mathsf{LK\ddot{u}rz}\!\left(A,\star\right)}
  \DeltaPrem{Rechtskürzbarkeit}{\mathsf{RK\ddot{u}rz}\!\left(A,\star\right)}
  \DeltaRow{\textbf{Neues Symbol}}{}
  \DeltaPrem{Kürzbarkeit}{\mathsf{K\ddot{u}rz}\!\left(A,\star\right)}
}

\FormulaAxiomDeltaK[Eine kürzbare Operation ist linkskürzbar]{%
  \mathsf{K\ddot{u}rz}\!\left(A,\star\right)
  \vdash\mathsf{LK\ddot{u}rz}\!\left(A,\star\right)%
}{CancellativeLeftAxiom}{%
  \DeltaRow{Mengen}{A}
}

\FormulaAxiomDeltaK[Eine kürzbare Operation ist rechtskürzbar]{%
  \mathsf{K\ddot{u}rz}\!\left(A,\star\right)
  \vdash\mathsf{RK\ddot{u}rz}\!\left(A,\star\right)%
}{CancellativeRightAxiom}{%
  \DeltaRow{Mengen}{A}
}

\FormulaDefDeltaK[Begriff der kürzbaren Halbgruppe]{%
  \CancellativeSemigroupAlg{A}{\star}%
}{CancellativeSemigroupDef}{%
  \DeltaRow{Mengen}{A}
  \DeltaRow{\textbf{Axiome}}{}
  \DeltaPrem{Linkskürzbare Halbgruppe}{%
    \LeftCancellativeSemigroupAlg{A}{\star}}
  \DeltaPrem{Rechtskürzbare Halbgruppe}{%
    \RightCancellativeSemigroupAlg{A}{\star}}
  \DeltaRow{\textbf{Neues Symbol}}{}
  \DeltaPrem{Kürzbare Halbgruppe}{%
    \CancellativeSemigroupAlg{A}{\star}}
}

\FormulaAxiomDeltaK[Zugriff auf die linkskürzbare Halbgruppe]{%
  \CancellativeSemigroupAlg{A}{\star}
  \vdash\LeftCancellativeSemigroupAlg{A}{\star}%
}{CancellativeSemigroupLeftBaseAxiom}{%
  \DeltaRow{Mengen}{A}
}

\FormulaAxiomDeltaK[Zugriff auf die rechtskürzbare Halbgruppe]{%
  \CancellativeSemigroupAlg{A}{\star}
  \vdash\RightCancellativeSemigroupAlg{A}{\star}%
}{CancellativeSemigroupRightBaseAxiom}{%
  \DeltaRow{Mengen}{A}
}

\FormulaAxiomDeltaK[Zugriff auf die Halbgruppe]{%
  \CancellativeSemigroupAlg{A}{\star}
  \vdash\SemigroupAlg{A}{\star}%
}{CancellativeSemigroupBaseAxiom}{%
  \DeltaRow{Mengen}{A}
}

\FormulaAxiomDeltaK[Zugriff auf die kürzbare Operation]{%
  \CancellativeSemigroupAlg{A}{\star}
  \vdash\mathsf{K\ddot{u}rz}\!\left(A,\star\right)%
}{CancellativeSemigroupPropertyAxiom}{%
  \DeltaRow{Mengen}{A}
}

\FormulaAxiomDeltaK[Direkte Linkskürzung]{%
  \CancellativeSemigroupAlg{A}{\star}\dsep
  a,b,c\in A\dsep a\star b=a\star c
  \vdash b=c%
}{CancellativeSemigroupLeftCancellationAxiom}{%
  \DeltaRow{Mengen}{A\dsep a\dsep b\dsep c}
}

\FormulaAxiomDeltaK[Direkte Rechtskürzung]{%
  \CancellativeSemigroupAlg{A}{\star}\dsep
  a,b,c\in A\dsep b\star a=c\star a
  \vdash b=c%
}{CancellativeSemigroupRightCancellationAxiom}{%
  \DeltaRow{Mengen}{A\dsep a\dsep b\dsep c}
}
